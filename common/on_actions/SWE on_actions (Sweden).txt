on_actions = {

	on_startup = {
		effect = {
			if = {
				limit = { has_start_date < 1936.01.02 }
				SWE = {
					country_event = { id = NW_SWE.8 days = 260 }
					country_event = { id = NW_SWE.17 days = 1026 }
					country_event = { id = NW_SWE.18 days = 314 }
					country_event = { id = NW_SWE.19 days = 648 }
					country_event = { id = NW_SWE.20 days = 496 }
					country_event = { id = NW_SWE.31 days = 225 }
					country_event = { id = NW_SWE.37 days = 1675 }
					country_event = { id = NW_SWE.41 days = 683 }
					news_event = { id = NW_SWE_news.3 days = 300 }
				}
				GER = { 
					country_event = { id = NW_SWE.22 days = 1744 }
				
				}
			}
		}
	}
	
	on_monthly = { 
		effect = {
			if = {
				limit = { has_start_date < 1936.01.02 }
				SWE = {
					if = { 
						limit = { 
							threat > 0.25 
							NOT = { has_country_flag = SWE_tension_flag_25 }
						}
						country_event = NW_SWE.25
					}
					else_if = { 
						limit = { 
							threat > 0.5 
							NOT = { has_country_flag = SWE_tension_flag_50 }
						}
						country_event = NW_SWE.25
					}
					else_if = { 
						limit = { 
							threat > 0.75 
							NOT = { has_country_flag = SWE_tension_flag_75 }
						}
						country_event = NW_SWE.25
					}
					else_if = { 
						limit = { 
							threat > 0.95 
							NOT = { has_country_flag = SWE_tension_flag_100 }
						}
						country_event = NW_SWE.25
					}
				}
			}
		}
	}
	
	on_declare_war = {
		effect = {
			if = {
				limit = {
					FROM = { tag = FIN }
					ROOT = { tag = SOV }
					SWE = { NOT = { has_idea = SWE_activism } }
				}
				SWE = { country_event = NW_SWE.26 }
				if = { 
					limit = { FIN = { NOT = { has_country_flag = FIN_was_attacked_by_ussr } } }
					FIN = { set_country_flag = FIN_was_attacked_by_ussr }
				}
			}
            if = {
                limit = {
                    ROOT = { tag = GER }
					GER = { NOT = { has_country_flag = GER_has_declared_an_offensive_war } }
                }
                GER = { set_country_flag = GER_has_declared_an_offensive_war }
			}
			if = { 
				limit = {
					NOR = { NOT = { has_country_flag = NOR_was_attacked } }
					FROM = { tag = NOR }
				}
				NOR = { set_country_flag = NOR_was_attacked }
			}
		}	
	}
}
