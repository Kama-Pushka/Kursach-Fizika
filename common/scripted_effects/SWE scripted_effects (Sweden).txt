	set_temp_variable = { var = SWE_temp_deputies value = SWE_popularity_of_General_Electoral_League }
	divide_temp_variable = { var = SWE_temp_deputies value = 100 }
	multiply_temp_variable = { var = SWE_temp_deputies value = 30 }
	round_temp_variable = SWE_temp_deputies
	add_to_variable = { var = SWE_deputies_of_General_Electoral_League value = SWE_temp_deputies }
	
	set_temp_variable = { var = SWE_temp_deputies value = SWE_popularity_of_nazi }
	divide_temp_variable = { var = SWE_temp_deputies value = 100 }
	multiply_temp_variable = { var = SWE_temp_deputies value = 30 }
	round_temp_variable = SWE_temp_deputies
	add_to_variable = { var = SWE_deputies_of_nazi value = SWE_temp_deputies }
	
	# проверка чтобы было точно 230
	
	set_temp_variable = { var = SWE_summa value = SWE_deputies_of_KPSH }
	add_to_temp_variable = { var = SWE_summa value = SWE_deputies_of_socialistic_party }
	add_to_temp_variable = { var = SWE_summa value = SWE_deputies_of_SDRPSH }
	add_to_temp_variable = { var = SWE_summa value = SWE_deputies_of_Farm_league }
	add_to_temp_variable = { var = SWE_summa value = SWE_deputies_of_Peoples_party }
	add_to_temp_variable = { var = SWE_summa value = SWE_deputies_of_General_Electoral_League }
	add_to_temp_variable = { var = SWE_summa value = SWE_deputies_of_nazi }
	
	if = { 
		limit = { SWE_summa < 230 }
		set_temp_variable = { var = SWE_lack_of_dep value = 230 }
		set_temp_variable = { var = SWE_antisumm value = SWE_summa }
		multiply_temp_variable = { var = SWE_antisumm value = -1 }
		add_to_temp_variable = { var = SWE_lack_of_dep value = SWE_antisumm }
		if = { 
			limit = { SWE_deputies_of_General_Electoral_League < SWE_deputies_of_SDRPSH }
			add_to_variable = { var = SWE_deputies_of_SDRPSH value = SWE_lack_of_dep }
		}
		else = { 
			add_to_variable = { var = SWE_deputies_of_General_Electoral_League value = SWE_lack_of_dep }
		}
	}
	else_if = { 
		limit = { SWE_summa > 230 }
		set_temp_variable = { var = SWE_temp_delete value = SWE_summa }
		add_to_temp_variable = { var = SWE_temp_delete value = 230 }
		multiply_temp_variable = { var = SWE_temp_delete value = -1 }
		add_to_variable = { var = SWE_deputies_of_nazi value = SWE_temp_delete }
		if = { 
			limit = { SWE_deputies_of_nazi < 0 } 
			set_temp_variable = { var = SWE_temp_delete value = SWE_deputies_of_nazi }
			clamp_variable = { var = SWE_deputies_of_nazi min = 0 }
			add_to_variable = { var = SWE_deputies_of_KPSH value = SWE_temp_delete }
			if = { 
				limit = { SWE_deputies_of_KPSH < 0 } 
				set_temp_variable = { var = SWE_temp_delete value = SWE_deputies_of_KPSH }
				clamp_variable = { var = SWE_deputies_of_KPSH min = 0 }
				add_to_variable = { var = SWE_deputies_of_socialistic_party value = SWE_temp_delete }
				if = { 
					limit = { SWE_deputies_of_socialistic_party < 0 } 
					set_temp_variable = { var = SWE_temp_delete value = SWE_deputies_of_socialistic_party }
					clamp_variable = { var = SWE_deputies_of_socialistic_party min = 0 }
					add_to_variable = { var = SWE_deputies_of_Peoples_party value = SWE_temp_delete }
				}
			}
		}
	}
	
}
